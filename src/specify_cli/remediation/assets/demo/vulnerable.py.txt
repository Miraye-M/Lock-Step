from pathlib import Path


class DownloadRoot:
    def __init__(self, root):
        self.root = Path(root)

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()

    def close(self):
        pass

    def read(self, user_path):
        return (self.root / user_path).read_bytes()
